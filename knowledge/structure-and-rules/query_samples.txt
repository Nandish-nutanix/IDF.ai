# group limit, raw limit and group sort and group limit. raw_column should
@ always be present for an attribute on which we are performing sorting.
query {
  entity_list {
    entity_type_name: "vm"
  }
  group_by {
    group_by_column: "hypervisor_cpu_usage_ppm"
    group_by_column_type: kContinuous
    group_sort_order {
      sort_column: "hypervisor_cpu_usage_ppm"
    }
    raw_sort_order {
      sort_column: "num_vcpus"
    }
    raw_columns {
      column: "num_vcpus"
    }
    group_limit {
      limit: 10
      offset: 0
    }
    raw_limit {
      limit: 2
      offset: 10
    }
    num_buckets: 4
    bucket_boundary: 150
  }
  query_name: "group_and_raw_limit_and_offset"
}

#To return total entity count in query response, set flags=2.
#The query only works for non-evictable type entities and wildcard
#queries.
#The flag will only return the entity count without any raw results
#or guids. If raw columns ae specified in the query arg, the flag
#would be deemed invalid.
query {
      entity_list {
        entity_type_name: "vm"
      }
      flags: 2
      query_name: "entity_count_query"
    }

#where clause can be nested using the boolean operators. Example : kAnd,
#kOr, kNot
query = '''
query {
  entity_list {
    entity_type_name: "vm"
  }
  where_clause {
    lhs { 
      comparison_expr {
        lhs {
          leaf {
            column: "node"
          }
        }
        operator: kNE
        rhs {
          leaf {
            value {
              str_value: "node_1"
            }
          }
        }
      }
    }
    operator: kAnd
    rhs {
      comparison_expr {
        lhs {
          leaf {
            column: "vm_name"
          }
        }
        operator: kExists   
      }
    }
  }
  query_name: "nested_where_clause"
